function atualizarAplicativo() { let totalEntradas = 0, totalSaidas = 0, totalCartao = 0; const filtroMes = document.getElementById('filtro-mes').value; transacoes.forEach(t => { if (filtroMes !== 'todos') { const [m, a] = filtroMes.split('-'); if (t.mes != m || t.ano != a) return; } if (t.tipo === 'entrada') totalEntradas += t.valor; if (t.tipo === 'saida') totalSaidas += t.valor; if (t.tipo === 'cartao') totalCartao += t.valor; }); const saldoFinal = totalEntradas - totalSaidas - totalCartao; const totalGastos = totalSaidas + totalCartao; // CORREÇÃO: Média diária real baseada em base fixa de 30 dias conforme sua preferência const mediaGastoDiario = totalGastos / 30; document.getElementById('card-entradas').setAttribute('data-valor', totalEntradas.toFixed(2)); document.getElementById('card-saidas').setAttribute('data-valor', totalSaidas.toFixed(2)); document.getElementById('card-cartao').setAttribute('data-valor', totalCartao.toFixed(2)); document.getElementById('card-saldo').setAttribute('data-valor', saldoFinal.toFixed(2)); // Atualiza o valor do elemento document.getElementById('previsao-valor').setAttribute('data-valor', mediaGastoDiario.toFixed(2)); // CONTROLE DA SUGESTÃO DE RESERVA SE SALDO FOR POSITIVO const areaReserva = document.getElementById('card-reserva'); if (saldoFinal > 0) { const valorParaGuardar = saldoFinal * (porcentagemReserva / 100); document.getElementById('reserva-valor').setAttribute('data-valor', valorParaGuardar.toFixed(2)); document.getElementById('reserva-porcentagem-label').innerText = porcentagemReserva; areaReserva.style.display = 'flex'; } else { areaReserva.style.display = 'none'; } // CONTROLE DE ALERTA DE ORÇAMENTO COMPROMETIDO const alertaComprometido = document.getElementById('card-alerta-comprometido'); if (totalEntradas > 0 && (totalGastos / totalEntradas) >= 0.80) { alertaComprometido.style.display = 'block'; } else { alertaComprometido.style.display = 'none'; } const cardSaldo = document.getElementById('card-saldo'); cardSaldo.style.color = saldoFinal >= 0 ? 'var(--azul-saldo)' : 'var(--saida)'; atualizarVisualValores(); renderizarHistorico(); }
